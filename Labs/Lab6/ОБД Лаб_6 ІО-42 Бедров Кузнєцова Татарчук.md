#### Міграційні скрипти у папці src/main/resources/db/migration/

V2__add_date_to_release.sql
```
ALTER TABLE release_ ADD COLUMN date_ date;
```
V3__delete_date_from_review.sql
```
ALTER TABLE review DROP COLUMN review_date;
```
V4__create_investment_table.sql
```
CREATE TABLE investment ( 
id int PRIMARY KEY, name varchar(70) not NULL, 
amount int NOT NULL, 
date_ date NOT NULL, 
project int, 

CONSTRAINT fk_project 
FOREIGN KEY (project) 
REFERENCES project(project_id) 
on DELETE SET NULL 
on UPDATE CASCADE );
```
